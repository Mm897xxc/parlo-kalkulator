 * Zalijepi CIJELI ovaj tekst na KRAJ functions.php (child tema).
 * NE dodaji "<?php" na početak – functions.php ga već ima.
 *
 * Korištenje: na stranici (npr. naslovnica) dodaj blok/sekciju
 * "Shortcode" i upiši:
 *     [eshopper_carousel]
 * Opcije:
 *     [eshopper_carousel limit="8" featured="1" naslov="Usporedi cijene prije kupnje"]
 *     featured="1" = prikaže i okvir "Primjer usporedbe" (proizvod s najvećom razlikom u cijeni)
 *
 * Podaci: https://api.getparlo.io/catalog/highlights
 * Dohvaća se na serveru i sprema u cache 15 min; ako API padne,
 * prikazuje se zadnja uspješno dohvaćena verzija.
 * Prikazuje se samo na glavnoj (hrvatskoj) stranici.
 * ==================================================================== */

if ( ! function_exists( 'ekp_esh_highlights' ) ) {
    function ekp_esh_highlights() {
        $cached = get_transient( 'ekp_esh_highlights' );
        if ( is_array( $cached ) ) {
            return $cached;
        }
        $data = null;
        $resp = wp_remote_get( 'https://api.getparlo.io/catalog/highlights', array( 'timeout' => 5 ) );
        if ( ! is_wp_error( $resp ) && 200 === (int) wp_remote_retrieve_response_code( $resp ) ) {
            $json = json_decode( wp_remote_retrieve_body( $resp ), true );
            if ( is_array( $json ) && ! empty( $json['data'] ) && is_array( $json['data'] ) ) {
                $data = $json;
                update_option( 'ekp_esh_highlights_last', $json, false );
            }
        }
        if ( ! $data ) {
            $data = get_option( 'ekp_esh_highlights_last', array() ); // zadnja dobra verzija
        }
        set_transient( 'ekp_esh_highlights', is_array( $data ) ? $data : array(), 15 * MINUTE_IN_SECONDS );
        return is_array( $data ) ? $data : array();
    }
}

if ( ! function_exists( 'ekp_esh_price' ) ) {
    function ekp_esh_price( $p ) {
        return number_format( (float) $p, 2, ',', '.' ) . ' €';
    }
}

if ( ! function_exists( 'ekp_esh_trgovine' ) ) {
    // 1 trgovini, 2-4 trgovine, 5+ trgovina (11-14 trgovina, 21 trgovini...)
    function ekp_esh_trgovine( $n ) {
        $n = (int) $n;
        $d = $n % 10;
        $s = $n % 100;
        if ( 1 === $d && 11 !== $s ) {
            return 'u ' . $n . ' trgovini';
        }
        if ( $d >= 2 && $d <= 4 && ( $s < 12 || $s > 14 ) ) {
            return 'u ' . $n . ' trgovine';
        }
        return 'u ' . $n . ' trgovina';
    }
}

if ( ! function_exists( 'ekp_esh_url' ) ) {
    function ekp_esh_url( $path ) {
        $path = (string) $path;
        if ( '' === $path || '/' !== $path[0] ) {
            return '';
        }
        return 'https://eshopper.hr' . $path . ( false === strpos( $path, '?' ) ? '?' : '&' ) . 'utm_source=ekuponi&utm_medium=carousel';
    }
}

if ( ! function_exists( 'ekp_esh_carousel_shortcode' ) ) {
    function ekp_esh_carousel_shortcode( $atts ) {
        if ( is_multisite() && ! is_main_site() ) {
            return '';
        }
        $a = shortcode_atts( array(
            'limit'    => 12,
            'featured' => 0,
            'naslov'   => 'Usporedi cijene prije kupnje',
        ), $atts, 'eshopper_carousel' );

        $json  = ekp_esh_highlights();
        $items = isset( $json['data'] ) && is_array( $json['data'] ) ? $json['data'] : array();
        if ( ! $items ) {
            return '';
        }

        static $n = 0;
        $n++;
        $uid = 'ekp-esh-car-' . $n;

        $f       = isset( $json['featured'] ) && is_array( $json['featured'] ) ? $json['featured'] : null;
        $skip_id = ( $a['featured'] && $f && ! empty( $f['id'] ) ) ? $f['id'] : '';

        $cards = '';
        $shown = 0;
        foreach ( $items as $it ) {
            if ( $shown >= (int) $a['limit'] ) {
                break;
            }
            if ( $skip_id && isset( $it['id'] ) && $it['id'] === $skip_id ) {
                continue; // već je u okviru "Primjer usporedbe"
            }
            $url   = ekp_esh_url( isset( $it['path'] ) ? $it['path'] : '' );
            $title = isset( $it['title'] ) ? trim( $it['title'] ) : '';
            $img   = isset( $it['image_url'] ) ? $it['image_url'] : '';
            // preskoči proizvode bez linka, slike ili s prekratkim nazivom (npr. samo "5030")
            if ( ! $url || ! $img || mb_strlen( $title ) < 8 || empty( $it['min_price'] ) ) {
                continue;
            }
            $brand = isset( $it['brand'] ) ? trim( $it['brand'] ) : '';
            $cnt   = isset( $it['merchants_count'] ) ? (int) $it['merchants_count'] : 0;

            $cards .= '<a class="ekp-esh-card" href="' . esc_url( $url ) . '" target="_blank" rel="noopener">'
                . '<span class="ekp-esh-img"><img src="' . esc_url( $img ) . '" alt="' . esc_attr( $title ) . '" width="160" height="160" loading="lazy" decoding="async" referrerpolicy="no-referrer"></span>'
                . ( $brand ? '<span class="ekp-esh-brand">' . esc_html( $brand ) . '</span>' : '' )
                . '<span class="ekp-esh-title">' . esc_html( $title ) . '</span>'
                . '<span class="ekp-esh-price">od ' . esc_html( ekp_esh_price( $it['min_price'] ) ) . '</span>'
                . ( $cnt > 1 ? '<span class="ekp-esh-cnt">' . esc_html( ekp_esh_trgovine( $cnt ) ) . '</span>' : '' )
                . '</a>';
            $shown++;
        }
        if ( ! $cards ) {
            return '';
        }

        // Okvir "Primjer usporedbe" (neobavezno)
        $feat_html = '';
        if ( $a['featured'] && is_array( $f ) && ! empty( $f['offers'] ) && ekp_esh_url( isset( $f['path'] ) ? $f['path'] : '' ) ) {
            $max  = 0;
            foreach ( $f['offers'] as $o ) {
                $max = max( $max, (float) $o['price'] );
            }
            $rows = '';
            $i    = 0;
            foreach ( $f['offers'] as $o ) {
                $w     = $max > 0 ? max( 8, round( 100 * (float) $o['price'] / $max ) ) : 100;
                $rows .= '<li' . ( 0 === $i ? ' class="best"' : '' ) . '><span>' . esc_html( $o['merchant'] ) . '</span><b>' . esc_html( ekp_esh_price( $o['price'] ) ) . '</b><i style="width:' . (int) $w . '%"></i></li>';
                $i++;
            }
            $feat_html = '<div class="ekp-esh-feat"><span class="ekp-esh-feat-tag">Primjer usporedbe</span>'
                . '<a class="ekp-esh-feat-head" href="' . esc_url( ekp_esh_url( $f['path'] ) ) . '" target="_blank" rel="noopener">'
                . ( ! empty( $f['image_url'] ) ? '<img src="' . esc_url( $f['image_url'] ) . '" alt="' . esc_attr( $f['title'] ) . '" width="64" height="64" loading="lazy" referrerpolicy="no-referrer">' : '' )
                . '<span>' . ( ! empty( $f['brand'] ) ? '<small>' . esc_html( $f['brand'] ) . '</small>' : '' ) . esc_html( $f['title'] ) . '</span></a>'
                . '<ul>' . $rows . '</ul>'
                . '<a class="ekp-esh-feat-more" href="' . esc_url( ekp_esh_url( $f['path'] ) ) . '" target="_blank" rel="noopener">Usporedi ' . esc_html( ekp_esh_trgovine( count( $f['offers'] ) ) ) . ' →</a></div>';
        }

        ob_start();
        ?>
