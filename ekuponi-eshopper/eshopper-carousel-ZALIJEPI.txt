 * eShopper carousel za eKupone (u stilu sekcije "Istaknute trgovine")
 * Zalijepi CIJELI ovaj tekst na KRAJ functions.php (child tema),
 * umjesto prethodne verzije. NE dodaji "<?php" na početak.
 *
 * Korištenje (Shortcode blok/sekcija):
 *     [eshopper_carousel]
 *     [eshopper_carousel limit="10" naslov="Usporedi cijene"]
 *
 * Podaci: https://api.getparlo.io/catalog/highlights
 * Dohvat na serveru, cache 15 min; ako API padne, prikazuje se
 * zadnja uspješna verzija. Samo na glavnoj (hrvatskoj) stranici.
 * ==================================================================== */

if ( ! function_exists( 'ekp_esh_highlights' ) ) {
    function ekp_esh_highlights() {
        $cached = get_transient( 'ekp_esh_highlights' );
        if ( is_array( $cached ) ) {
            return $cached;
        }
        $data = null;
        $resp = wp_remote_get( 'https://api.getparlo.io/catalog/highlights', array( 'timeout' => 5 ) );
        if ( ! is_wp_error( $resp ) && 200 === (int) wp_remote_retrieve_response_code( $resp ) ) {
            $json = json_decode( wp_remote_retrieve_body( $resp ), true );
            if ( is_array( $json ) && ! empty( $json['data'] ) && is_array( $json['data'] ) ) {
                $data = $json;
                update_option( 'ekp_esh_highlights_last', $json, false );
            }
        }
        if ( ! $data ) {
            $data = get_option( 'ekp_esh_highlights_last', array() );
        }
        set_transient( 'ekp_esh_highlights', is_array( $data ) ? $data : array(), 15 * MINUTE_IN_SECONDS );
        return is_array( $data ) ? $data : array();
    }
}

if ( ! function_exists( 'ekp_esh_price' ) ) {
    function ekp_esh_price( $p ) {
        return number_format( (float) $p, 2, ',', '.' ) . ' €';
    }
}

if ( ! function_exists( 'ekp_esh_trgovine' ) ) {
    // 1 trgovina, 2-4 trgovine, 5+ trgovina
    function ekp_esh_trgovine( $n ) {
        $n = (int) $n;
        $d = $n % 10;
        $s = $n % 100;
        if ( $d >= 2 && $d <= 4 && ( $s < 12 || $s > 14 ) ) {
            return $n . ' trgovine';
        }
        if ( 1 === $d && 11 !== $s ) {
            return $n . ' trgovina';
        }
        return $n . ' trgovina';
    }
}

if ( ! function_exists( 'ekp_esh_url' ) ) {
    function ekp_esh_url( $path ) {
        $path = (string) $path;
        if ( '' === $path || '/' !== $path[0] ) {
            return '';
        }
        return 'https://eshopper.hr' . $path . ( false === strpos( $path, '?' ) ? '?' : '&' ) . 'utm_source=ekuponi&utm_medium=carousel';
    }
}

if ( ! function_exists( 'ekp_esh_carousel_shortcode' ) ) {
    function ekp_esh_carousel_shortcode( $atts ) {
        if ( is_multisite() && ! is_main_site() ) {
            return '';
        }
        $a = shortcode_atts( array(
            'limit'  => 12,
            'naslov' => 'Usporedi cijene',
        ), $atts, 'eshopper_carousel' );

        $json  = ekp_esh_highlights();
        $items = isset( $json['data'] ) && is_array( $json['data'] ) ? $json['data'] : array();

        $cards = '';
        $shown = 0;
        foreach ( $items as $it ) {
            if ( $shown >= (int) $a['limit'] ) {
                break;
            }
            $url   = ekp_esh_url( isset( $it['path'] ) ? $it['path'] : '' );
            $title = isset( $it['title'] ) ? trim( $it['title'] ) : '';
            $img   = isset( $it['image_url'] ) ? $it['image_url'] : '';
            if ( ! $url || ! $img || mb_strlen( $title ) < 8 || empty( $it['min_price'] ) ) {
                continue; // bez linka, slike ili s neispravnim nazivom
            }
            $cnt = isset( $it['merchants_count'] ) ? (int) $it['merchants_count'] : 0;

            $cards .= '<a class="ekp-esh-card" href="' . esc_url( $url ) . '" target="_blank" rel="noopener" title="' . esc_attr( $title ) . '">'
                . '<span class="ekp-esh-img"><img src="' . esc_url( $img ) . '" alt="' . esc_attr( $title ) . '" width="130" height="110" loading="lazy" decoding="async" referrerpolicy="no-referrer"></span>'
                . '<span class="ekp-esh-info">'
                . '<span class="ekp-esh-title">' . esc_html( $title ) . '</span>'
                . '<span class="ekp-esh-price">od ' . esc_html( ekp_esh_price( $it['min_price'] ) ) . '</span>'
                . ( $cnt > 1 ? '<span class="ekp-esh-cnt">' . esc_html( ekp_esh_trgovine( $cnt ) ) . '</span>' : '' )
                . '</span></a>';
            $shown++;
        }
        if ( ! $cards ) {
            return '';
        }

        static $n = 0;
        $n++;
        $uid = 'ekp-esh-car-' . $n;

        ob_start();
        ?>
